  function currentMonthKey(){return new Date().toISOString().slice(0,7)}
  function dayDiff(date){const a=D(today()),b=D(date);return b?Math.ceil((b-a)/86400000):99999}
  function hashId(s){let h=0;for(let i=0;i<String(s).length;i++)h=((h<<5)-h)+String(s).charCodeAt(i)|0;return Math.abs(h)||1}

  function scheduleUpcomingNotifications(){
    if(!state.settings?.notifications || !window.AndroidBridge || typeof window.AndroidBridge.scheduleNotification!=="function")return;
    window.OZ_V73.deriveCalendar(state).forEach(e=>{
      const diff=dayDiff(e.date);if(diff<0||diff>30)return;
      const dt=new Date(e.date+"T09:00:00");dt.setDate(dt.getDate()-1);
      if(dt.getTime()<Date.now()+60000)dt.setTime(Date.now()+60000);
      try{window.AndroidBridge.scheduleNotification(hashId(e.id||e.title+e.date),"ÖZBİLTAR · "+(e.type||"Hatırlatıcı"),e.title,dt.getTime())}catch(err){console.warn(err)}
    });
  }

  function renderCalendar(){
    const all=window.OZ_V73.deriveCalendar(state),horizon=N($("calendarHorizon")?.value||30),up=all.filter(e=>{const d=dayDiff(e.date);return d>=0&&d<=horizon}),over=all.filter(e=>dayDiff(e.date)<0);
    $("calendarStats").innerHTML=[["Bugün",all.filter(e=>e.date===today()).length],["7 Gün",all.filter(e=>{const d=dayDiff(e.date);return d>=0&&d<=7}).length],["Geciken",over.length],["Toplam",all.length]].map(v=>`<div class="mini-kpi"><span>${v[0]}</span><strong>${v[1]}</strong></div>`).join("");
